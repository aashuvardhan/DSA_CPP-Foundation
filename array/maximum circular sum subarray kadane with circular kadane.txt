
    int kadane(vector<int>& arr){
        int maxhere=arr[0];
        int maxtill=arr[0];

        for(int i=1;i<arr.size();i++){
            maxhere=max(maxhere+arr[i],arr[i]);
            maxtill=max(maxtill,maxhere);
        }
        return maxtill;
    }


    int maxSubarraySumCircular(vector<int>& nums) {
        
        int normal=kadane(nums);
        
        if(normal<0){
            //if normal is less than 0 then it is sure that the entire array is already negative 
            //hence if you go finding the circular => sum(of all neg)+sum(of all pos) = 0
            //which will denote empty subarray. which is not the requirement. 
            return normal;
        }


        int total=0;
        for(int i=0;i<nums.size();i++){
            total+=nums[i];
            nums[i]=-nums[i];
        }

        int circular=total+kadane(nums);
        return max(normal,circular);


    }
